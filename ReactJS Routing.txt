Step 3: Create Employee List Component
src/components/EmployeeList.jsx
import { useEffect, useState } from "react";
import { getEmployees } from "../services/employeeService";

function EmployeeList() {

    const [employees, setEmployees] = useState([]);
    const [loading, setLoading] = useState(true);
    const [error, setError] = useState("");

    useEffect(() => {

        getEmployees()
            .then((response) => {
                setEmployees(response.data);
            })
            .catch(() => {
                setError("Unable to fetch employee data.");
            })
            .finally(() => {
                setLoading(false);
            });

    }, []);

    if (loading) {
        return <h2>Loading Employees...</h2>;
    }

    if (error) {
        return <h2>{error}</h2>;
    }

    return (
        <div className="container">
            <h2>Employee List</h2>

            <table border="1" cellPadding="10">

                <thead>
                    <tr>
                        <th>ID</th>
                        <th>Name</th>
                        <th>Department</th>
                        <th>Salary</th>
                    </tr>
                </thead>

                <tbody>

                    {employees.map((emp) => (

                        <tr key={emp.id}>
                            <td>{emp.id}</td>
                            <td>{emp.name}</td>
                            <td>{emp.department}</td>
                            <td>{emp.salary}</td>
                        </tr>

                    ))}

                </tbody>

            </table>

        </div>
    );
}

export default EmployeeList;

Module 1: What is React Router?
React Router is a library used for navigation in React applications.

Instead of loading a new HTML page, React changes only the required component, making the application faster.

Without React Router
Home.html
Employee.html
About.html
Contact.html
Every click reloads the entire page.

With React Router
Home Component
Employee Component
About Component
Contact Component
Only the component changes without reloading the page.

Module 2: What is SPA (Single Page Application)?
A Single Page Application (SPA) loads one HTML page initially and updates only the content when users navigate.

Traditional Website
Browser

↓

Home.html

↓

Employee.html

↓

About.html

↓

Contact.html
Every navigation reloads the page.

React SPA
Browser

↓

index.html

↓

Home Component

↓

Employee Component

↓

About Component

↓

Contact Component
Only components change.

Advantages of SPA
Faster navigation

Better user experience

No full page refresh

Less server load

Smooth transitions

Module 2: Controlled Components
A Controlled Component is a form element whose value is controlled by React state.

Example
import { useState } from "react";

function EmployeeForm() {

    const [name, setName] = useState("");

    return (
        <>
            <input
                type="text"
                value={name}
                onChange={(e) => setName(e.target.value)}
            />

            <p>Name : {name}</p>
        </>
    );
}

export default EmployeeForm;
Explanation
value={name} → Input value comes from React state.

onChange → Updates state whenever the user types.

setName() → Triggers a re-render with the latest value.

Module 3: Handling Multiple Inputs
import { useState } from "react";

function EmployeeForm() {

    const [employee, setEmployee] = useState({
        name: "",
        email: "",
        department: "",
        salary: ""
    });

    const handleChange = (event) => {

        const { name, value } = event.target;

        setEmployee({
            ...employee,
            [name]: value
        });

    };

    return (
        <>
            <input
                type="text"
                name="name"
                placeholder="Enter Name"
                value={employee.name}
                onChange={handleChange}
            />

            <input
                type="email"
                name="email"
                placeholder="Enter Email"
                value={employee.email}
                onChange={handleChange}
            />
        </>
    );
}

export default EmployeeForm;

Module 6: Error Messages
const [errors, setErrors] = useState({});
Validation Function

const validate = () => {

    let validationErrors = {};

    if (!employee.name.trim()) {
        validationErrors.name = "Name is required";
    }

    if (!employee.email.trim()) {
        validationErrors.email = "Email is required";
    }

    if (employee.salary <= 0) {
        validationErrors.salary = "Salary must be greater than zero";
    }

    setErrors(validationErrors);

    return Object.keys(validationErrors).length === 0;
};
