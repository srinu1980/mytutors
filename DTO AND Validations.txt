Correct. In the previous example, the Product Entity and DTOs contain almost the same fields, so beginners may not clearly understand why DTOs are needed.

In real-world corporate projects, Entities often contain database-related fields, audit fields, and sensitive information that should never be exposed to clients.

Now the purpose of DTO is very clear:

Entity (Database Layer)	DTO (API Layer)
Contains all database fields	Contains only required fields
May contain sensitive data	Hides sensitive data
Used by JPA/Hibernate	Used by Client/API
Represents Database Structure	Represents API Contract
Can change frequently	Usually stable
Corporate Interview Answer

Why do we use DTO in Spring Boot?

DTO (Data Transfer Object) is used to transfer data between the client and server without exposing the Entity directly. It helps hide sensitive information, perform request validation, reduce unnecessary data transfer, and maintain a clean separation between the API layer and database layer. In enterprise applications, DTOs improve security, maintainability, and API flexibility.



This is a good real-time example because password is a sensitive field. DTOs clearly demonstrate why we should not expose Entity objects directly.

package com.example.usermanagement.dto;

import jakarta.validation.constraints.Email;
import jakarta.validation.constraints.NotBlank;
import jakarta.validation.constraints.Pattern;
import jakarta.validation.constraints.Size;

import lombok.*;

@Getter
@Setter
@NoArgsConstructor
@AllArgsConstructor
public class UserRequestDTO {

    @NotBlank(message = "User Name is required")
    @Size(min = 3, max = 30,
          message = "User Name must contain 3 to 30 characters")
    private String userName;

    @NotBlank(message = "Email is required")
    @Email(message = "Enter valid email")
    private String email;

    @NotBlank(message = "Password is required")
    @Size(min = 8,
          message = "Password must contain minimum 8 characters")
    @Pattern(
            regexp = "^(?=.*[A-Z])(?=.*[a-z])(?=.*\\d).*$",
            message = "Password must contain uppercase, lowercase and number"
    )
    private String password;
}

5. Service Implementation
UserServiceImpl.java
package com.example.usermanagement.service;

import com.example.usermanagement.dto.*;
import com.example.usermanagement.entity.UserDetails;
import com.example.usermanagement.exception.ResourceNotFoundException;
import com.example.usermanagement.repository.UserRepository;

import lombok.RequiredArgsConstructor;

import org.springframework.stereotype.Service;

import java.util.List;

@Service
@RequiredArgsConstructor
public class UserServiceImpl
        implements UserService {

    private final UserRepository repository;

    @Override
    public UserResponseDTO saveUser(
            UserRequestDTO dto) {

        UserDetails user = UserDetails.builder()
                .userName(dto.getUserName())
                .email(dto.getEmail())
                .password(dto.getPassword())
                .build();

        UserDetails savedUser =
                repository.save(user);

        return mapToResponseDTO(savedUser);
    }

    @Override
    public UserResponseDTO getUserById(Long userId) {

        UserDetails user = repository.findById(userId)
                .orElseThrow(() ->
                        new ResourceNotFoundException(
                                "User Not Found : " + userId));

        return mapToResponseDTO(user);
    }

    @Override
    public List<UserResponseDTO> getAllUsers() {

        return repository.findAll()
                .stream()
                .map(this::mapToResponseDTO)
                .toList();
    }

    @Override
    public UserResponseDTO updateUser(
            Long userId,
            UserRequestDTO dto) {

        UserDetails user = repository.findById(userId)
                .orElseThrow(() ->
                        new ResourceNotFoundException(
                                "User Not Found : " + userId));

        user.setUserName(dto.getUserName());
        user.setEmail(dto.getEmail());
        user.setPassword(dto.getPassword());

        UserDetails updatedUser =
                repository.save(user);

        return mapToResponseDTO(updatedUser);
    }

    @Override
    public void deleteUser(Long userId) {

        UserDetails user = repository.findById(userId)
                .orElseThrow(() ->
                        new ResourceNotFoundException(
                                "User Not Found : " + userId));

        repository.delete(user);
    }

    private UserResponseDTO mapToResponseDTO(
            UserDetails user) {

        return UserResponseDTO.builder()
                .userId(user.getUserId())
                .userName(user.getUserName())
                .email(user.getEmail())
                .build();
    }
}
6. Controller Layer
UserController.java
package com.example.usermanagement.controller;

import com.example.usermanagement.dto.*;
import com.example.usermanagement.service.UserService;

import jakarta.validation.Valid;

import lombok.RequiredArgsConstructor;

import org.springframework.http.*;
import org.springframework.web.bind.annotation.*;

import java.util.List;

@RestController
@RequestMapping("/api/users")
@RequiredArgsConstructor
public class UserController {

    private final UserService service;

    @PostMapping
    public ResponseEntity<UserResponseDTO>
    saveUser(
            @Valid
            @RequestBody UserRequestDTO dto) {

        return new ResponseEntity<>(
                service.saveUser(dto),
                HttpStatus.CREATED);
    }

    @GetMapping("/{id}")
    public ResponseEntity<UserResponseDTO>
    getUserById(@PathVariable Long id) {

        return ResponseEntity.ok(
                service.getUserById(id));
    }

    @GetMapping
    public ResponseEntity<List<UserResponseDTO>>
    getAllUsers() {

        return ResponseEntity.ok(
                service.getAllUsers());
    }

    @PutMapping("/{id}")
    public ResponseEntity<UserResponseDTO>
    updateUser(
            @PathVariable Long id,
            @Valid @RequestBody UserRequestDTO dto) {

        return ResponseEntity.ok(
                service.updateUser(id, dto));
    }

    @DeleteMapping("/{id}")
    public ResponseEntity<String>
    deleteUser(@PathVariable Long id) {

        service.deleteUser(id);

        return ResponseEntity.ok(
                "User Deleted Successfully");
    }
}
